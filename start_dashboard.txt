@echo off
REM ============================================================================
REM Jindal Steel & Power - SMS Digital Transformation & Automation
REM In-Motion Weighbridge Dashboard - Windows Production Serving Script
REM ============================================================================

setlocal enabledelayedexpansion

TITLE IMWB Weighbridge Production Server

echo ============================================================================
echo Starting Jindal Steel In-Motion Weighbridge Production Server...
echo ============================================================================

REM Set environment variable defaults if not already defined in the environment
if "%IMWB_DATABASE%"=="" set IMWB_DATABASE=IMWB_STWB
if "%IMWB_TIMEZONE%"=="" set IMWB_TIMEZONE=Asia/Kolkata
if "%ODBC_DRIVER%"=="" set ODBC_DRIVER=ODBC Driver 18 for SQL Server
if "%PORT%"=="" set PORT=5000

echo Plant Local Timezone : %IMWB_TIMEZONE%
echo SQL Server Instance  : %IMWB_SERVER%
echo Target Database      : %IMWB_DATABASE%
echo HTTP Listening Port  : %PORT%

REM Check Python virtual environment
if exist "venv\Scripts\activate.bat" (
    echo Activating local virtual environment (venv)...
    call venv\Scripts\activate.bat
) else if exist "..\venv\Scripts\activate.bat" (
    echo Activating parent virtual environment (..\venv)...
    call ..\venv\Scripts\activate.bat
) else (
    echo [INFO] Running using system Python environment...
)

REM Verify waitress availability
python -c "import waitress" 2>nul
if %errorlevel% neq 0 (
    echo [WARNING] Waitress is not installed. Installing requirements...
    pip install -r requirements.txt
)

echo.
echo Serving production traffic on http://0.0.0.0:%PORT% via Waitress...
echo Press CTRL+C to terminate the server.
echo.

waitress-serve --listen=0.0.0.0:%PORT% app:app

pause
